(defvar-local my/bidi-direction 'right-to-left)

(defun my/set-rtl ()
  "Set buffer to RTL."
  (interactive)
  (setq-local bidi-paragraph-direction 'right-to-left)
  (setq-local my/bidi-direction 'right-to-left)
  (message "Direction: RTL"))

(defun my/set-ltr ()
  "Set buffer to LTR."
  (interactive)
  (setq-local bidi-paragraph-direction 'left-to-right)
  (setq-local my/bidi-direction 'left-to-right)
  (message "Direction: LTR"))

(defun my/toggle-direction ()
  "Toggle RTL/LTR."
  (interactive)
  (if (eq bidi-paragraph-direction 'right-to-left)
      (my/set-ltr)
    (my/set-rtl)))

(defun my/toggle-bidi-reordering ()
  "Toggle bidi reordering."
  (interactive)
  (setq bidi-display-reordering (not bidi-display-reordering))
  (message "Bidi: %s" (if bidi-display-reordering "ON" "OFF")))

(defun my/setup-rtl-mode ()
  "Setup RTL for prose."
  (setq-local bidi-paragraph-direction 'right-to-left))

(defun my/setup-ltr-mode ()

  (setq-local bidi-paragraph-direction 'left-to-right))

(defun my/kill-current-buffer ()
  (interactive)
  (kill-buffer (current-buffer)))

(defun my/kill-other-buffers ()
  (interactive)
  (let ((kept 0) (killed 0))
    (dolist (buf (buffer-list))
      (let ((name (buffer-name buf)))
        (cond
         ((eq buf (current-buffer)) (cl-incf kept))
         ((string-prefix-p " " name) (cl-incf kept))
         ((string-prefix-p "*" name) (cl-incf kept))
         (t (kill-buffer buf) (cl-incf killed)))))
    (message "Killed %d, kept %d" killed kept)))

(defun my/revert-buffer-no-confirm ()
  (interactive)
  (revert-buffer t t)
  (message "Reverted"))

        (progn

(defun seforim--filter-files (files)
  (let ((root (expand-file-name seforim-directory)))
    (seq-filter
     (lambda (f)
       (and (string-prefix-p root f)
            (member (downcase (or (file-name-extension f) "")) seforim-extensions)
            (file-exists-p f)))
     files)))

  (let ((root (expand-file-name seforim-directory)))

(defun seforim--search-recoll (query dir)
  (let* ((cmd (format "recoll -t -q %s dir:%s 2>/dev/null"
                      (shell-quote-argument query)
                      (shell-quote-argument dir)))
         (output (shell-command-to-string cmd))
         (candidates (seforim--parse-recoll output dir)))
    (if candidates
        (let* ((choice (completing-read (format "「%s」: " query) candidates nil t))
               (file (cdr (assoc choice candidates))))
          (when file
            (find-file file)
            (goto-char (point-min))
            (search-forward query nil t)))
      (user-error "No results: %s" query))))

(defun seforim--parse-recoll (output dir)
  (let ((results nil) (seen (make-hash-table :test 'equal)))
    (dolist (line (split-string output "\n" t))
      (when (string-match "\\(/[^[:cntrl:]]+\\)" line)
        (let* ((file (match-string 1 line))
               (ext (downcase (or (file-name-extension file) ""))))
          (when (and (member ext seforim-extensions)
                     (file-exists-p file)
                     (string-prefix-p dir (expand-file-name file))
                     (not (gethash file seen)))
            (puthash file t seen)
            (push (cons (file-relative-name file dir) file) results)))))
    (nreverse results)))

(defun seforim--number-to-hebrew (num)
  (let* ((ones '("" "א" "ב" "ג" "ד" "ה" "ו" "ז" "ח" "ט"))
         (tens '("" "י" "כ" "ל" "מ" "נ" "ס" "ע" "פ" "צ"))
         (hundreds '("" "ק" "ר" "ש" "ת"))
         (h (/ num 100)) (t-val (/ (mod num 100) 10)) (o (mod num 10)))
    (cond
     ((= (mod num 100) 15) (concat (nth h hundreds) "טו"))
     ((= (mod num 100) 16) (concat (nth h hundreds) "טז"))
     (t (concat (nth h hundreds) (nth t-val tens) (nth o ones))))))

  (unless (and buffer-file-name (string-match-p "Bavli" buffer-file-name))
    (user-error "Not in Bavli file"))
  (let* ((daf-num (read-number "Daf: " 2))
         (daf-side (completing-read "Side: " '("א" "ב") nil t))
         (daf-heading (format "דף %s ע\"%s" (seforim--number-to-hebrew daf-num) daf-side)))
    (goto-char (point-min))
    (if (re-search-forward (format "^\\*+ %s" (regexp-quote daf-heading)) nil t)
        (progn (org-reveal) (recenter))
      (user-error "Daf %s not found" daf-heading))))

** Documentation

*** Overview

This system provides rich content inside footnotes for:
- LaTeX (bigfoot + custom environments)
- Typst (native + helpers)
- ConTeXt (native + helpers)  
- Org-mode (preprocessor auto-converts on export)
- Pandoc (PDF, DOCX, HTML via ox-pandoc)

All support:
- 20 levels of headings
- 20 levels of nested lists (itemize, enumerate, description)
- Tables with colspan/rowspan, captions, numbering
- Code blocks
- Nested footnotes (10 levels)

*** Quick Reference

| System  | Rich Footnote Command/Block                    |
|---------+------------------------------------------------|
| LaTeX   | =\rfn{...}= or =\rfnB{...}= etc.               |
| Typst   | =#rfn[...]= or native =#footnote[...]=         |
| ConTeXt | =\startRFN= ... =\stopRFN=                     |
| Org     | Auto-detected, or =#+BEGIN_RICHFN id= block    |

**** Simple rich footnote:

(defun my/richfn--count-stars (str)
  "Count leading stars in STR."

(defun my/richfn--has-rich-content-p (text)
  "Check if TEXT contains rich org content."
  (and text
       (string-match-p
        (rx (or
             ;; Headings
             (seq bol (+ "*") " ")
             ;; Unordered lists
             (seq bol (* space) (any "-+*") " ")
             ;; Ordered lists
             (seq bol (* space) (+ digit) (any ".)") " ")
             ;; Description lists
             (seq bol (* space) "- " (* nonl) " :: ")
             ;; Tables
             (seq bol (* space) "|")
             ;; Code blocks
             (seq bol "#+begin_src")
             ;; Nested footnote references
             (seq "[fn:")))
        text)))

(defun my/richfn--get-footnote-definitions ()
  "Get all footnote definitions as alist of (id . (start end content))."
  (let ((footnotes nil))
    (save-excursion
      (goto-char (point-min))
      (while (re-search-forward "^\\[fn:\\([^]]+\\)\\][ \t]*" nil t)
        (let* ((id (match-string 1))
               (start (match-beginning 0))
               (content-start (match-end 0))
               (end (save-excursion
                      (if (re-search-forward "^\\[fn:\\|\\'" nil t)
                          (match-beginning 0)
                        (point-max))))
               (content (string-trim
                         (buffer-substring-no-properties content-start end))))
          (push (list id start end content) footnotes))))
    (nreverse footnotes)))

(defun my/richfn--parse-rich-blocks ()
  "Parse #+BEGIN_RICHFN blocks."
  (let ((blocks nil))
    (save-excursion
      (goto-char (point-min))
      (while (re-search-forward "^[ \t]*#\\+BEGIN_RICHFN[ \t]+\\([^ \t\n]+\\)" nil t)
        (let* ((id (match-string 1))
               (block-start (match-beginning 0))
               (content-start (1+ (line-end-position)))
               (block-end (when (re-search-forward "^[ \t]*#\\+END_RICHFN" nil t)
                            (match-end 0)))
               (content-end (match-beginning 0))
               (content (when block-end
                          (string-trim
                           (buffer-substring-no-properties content-start content-end)))))
          (when content
            (push (list id block-start block-end content) blocks)))))
    (nreverse blocks)))

(defun my/richfn--org-to-latex (text)
  "Convert org TEXT to LaTeX for footnote."
  (with-temp-buffer
    (insert text)
    
    ;; Convert headings (20 levels)
    (goto-char (point-min))
    (while (re-search-forward "^\\(\\*+\\) \\(.*\\)$" nil t)
      (let ((level (length (match-string 1)))
            (title (match-string 2)))
        (replace-match (format "\\\\fnh{%d}{%s}" level title) t t)))
    
    ;; Convert description lists first (before regular lists)
    (goto-char (point-min))
    (while (re-search-forward "^\\([ \t]*\\)- \\([^:]+\\) :: \\(.*\\)$" nil t)
      (let ((indent (match-string 1))
            (term (match-string 2))
            (desc (match-string 3)))
        (replace-match (format "%s\\\\item[%s] %s" indent term desc) t t)))
    
    ;; Track list state for wrapping
    (let ((in-itemize nil)
          (in-enumerate nil))
      
      ;; Convert unordered lists
      (goto-char (point-min))
      (while (re-search-forward "^\\([ \t]*\\)\\([-+*]\\) \\(.*\\)$" nil t)
        (let ((indent (match-string 1))
              (content (match-string 3)))
          (replace-match (format "%s\\\\item %s" indent content) t t)))
      
      ;; Convert ordered lists
      (goto-char (point-min))
      (while (re-search-forward "^\\([ \t]*\\)\\([0-9]+\\)[.)] \\(.*\\)$" nil t)
        (let ((indent (match-string 1))
              (content (match-string 3)))
          (replace-match (format "%s\\\\item %s" indent content) t t))))
    
    ;; Wrap consecutive \\item lines in environments
    (goto-char (point-min))
    (while (re-search-forward "\\(\\\\item.*\n\\)+" nil t)
      (let ((items (match-string 0))
            (start (match-beginning 0))
            (end (match-end 0)))
        (delete-region start end)
        (goto-char start)
        (insert "\\begin{fnitemize}\n" items "\\end{fnitemize}\n")))
    
    ;; Convert tables
    (goto-char (point-min))
    (when (re-search-forward "^[ \t]*|" nil t)
      (goto-char (point-min))
      (my/richfn--convert-tables-to-latex))
    
    ;; Convert code blocks
    (goto-char (point-min))
    (while (re-search-forward "^#\\+begin_src\\(?: \\([^ \t\n]+\\)\\)?[ \t]*\n\\(\\(?:.*\n\\)*?\\)#\\+end_src" nil t)
      (let ((lang (or (match-string 1) ""))
            (code (match-string 2)))
        (replace-match (format "\\\\begin{fncode}[%s]\n%s\\\\end{fncode}" lang code) t t)))
    
    ;; Convert inline code
    (goto-char (point-min))
    (while (re-search-forward "~\\([^~]+\\)~" nil t)
      (replace-match "\\\\fnc{\\1}" t nil))
    (goto-char (point-min))
    (while (re-search-forward "=\\([^=]+\\)=" nil t)
      (replace-match "\\\\fnc{\\1}" t nil))
    
    ;; Convert nested footnote references
    (goto-char (point-min))
    (while (re-search-forward "\\[fn:\\([^]]+\\)\\]" nil t)
      (let ((ref (match-string 1)))
        ;; Use next available level
        (replace-match (format "\\\\rfnmark{B}\\\\rfntext{B}{%s}" ref) t t)))
    
    (buffer-string)))

(defun my/richfn--convert-tables-to-latex ()
  "Convert org tables to LaTeX in current buffer."
  (let ((table-start nil)
        (table-lines nil)
        (col-count 0))
    (while (re-search-forward "^[ \t]*|\\(.*\\)|[ \t]*$" nil t)
      (let* ((line (match-string 1))
             (cells (split-string line "|"))
             (is-sep (string-match-p "^[-+]+$" (string-trim (car cells)))))
        (unless table-start
          (setq table-start (match-beginning 0)))
        (if is-sep
            (push 'separator table-lines)
          (setq col-count (max col-count (length cells)))
          (push cells table-lines))))
    
    (when table-lines
      (setq table-lines (nreverse table-lines))
      (let ((latex-table
             (concat
              (format "\\begin{fntab}{|%s|}\n\\hline\n"
                      (mapconcat (lambda (_) "l") (number-sequence 1 col-count) "|"))
              (mapconcat
               (lambda (row)
                 (if (eq row 'separator)
                     "\\hline"
                   (concat (mapconcat #'string-trim row " & ") " \\\\")))
               table-lines
               "\n")
              "\n\\hline\n\\end{fntab}")))
        (delete-region table-start (point))
        (goto-char table-start)
        (insert latex-table)))))

    (while (re-search-forward "^[ \t]*|\\(.*\\)|[ \t]*$" nil t)
      (let* ((line (match-string 1))
             (cells (split-string line "|"))
             (is-sep (string-match-p "^[-+]+$" (string-trim (car cells)))))
        (unless table-start
          (setq table-start (match-beginning 0)))
        (unless is-sep
          (push cells table-lines))))
    
    (when table-lines
      (setq table-lines (nreverse table-lines))
      (let ((html-table
             (concat
              "<table>\n"
              (mapconcat
               (lambda (row)
                 (concat "  <tr>"
                         (mapconcat
                          (lambda (cell) (format "<td>%s</td>" (string-trim cell)))
                          row
                          "")
                         "</tr>"))
               table-lines
               "\n")
              "\n</table>")))
        (delete-region table-start (point))
        (goto-char table-start)
        (insert html-table)))))

(defun my/richfn--get-converter (backend)
  "Get converter function for BACKEND."
  (pcase backend
    ((or 'latex 'beamer) #'my/richfn--org-to-latex)
    ('html #'my/richfn--org-to-html)
    ('pandoc #'my/richfn--org-to-latex)
    (_ #'my/richfn--org-to-latex)))

(defun my/richfn-preprocess-buffer (backend)
  "Preprocess rich footnotes before export to BACKEND."
  (my/richfn--log "Preprocessing for backend: %s" backend)
  
  (let ((converter (my/richfn--get-converter backend)))
    
    ;; Process explicit RICHFN blocks first
    (let ((blocks (my/richfn--parse-rich-blocks)))
      (my/richfn--log "Found %d RICHFN blocks" (length blocks))
      
      ;; Process in reverse to maintain positions
      (dolist (block (nreverse blocks))
        (let ((id (nth 0 block))
              (start (nth 1 block))
              (end (nth 2 block))
              (content (nth 3 block)))
          (my/richfn--log "Processing block: %s" id)
          (let ((converted (funcall converter content)))
            ;; Delete the block
            (delete-region start end)
            ;; Find or create footnote definition
            (save-excursion
              (goto-char (point-min))
              (if (re-search-forward (format "^\\[fn:%s\\]" (regexp-quote id)) nil t)
                  ;; Replace existing
                  (let ((def-start (match-beginning 0))
                        (def-end (save-excursion
                                   (if (re-search-forward "^\\[fn:\\|\\'" nil t)
                                       (match-beginning 0)
                                     (point-max)))))
                    (delete-region def-start def-end)
                    (goto-char def-start)
                    (insert (format "[fn:%s] %s\n" id converted)))
                ;; Create new at end
                (goto-char (point-max))
                (insert (format "\n[fn:%s] %s\n" id converted))))))))
    
    ;; Auto-detect rich content in remaining footnotes
    (let ((footnotes (my/richfn--get-footnote-definitions)))
      (my/richfn--log "Found %d footnotes to check" (length footnotes))
      
      ;; Process in reverse
      (dolist (fn (nreverse footnotes))
        (let ((id (nth 0 fn))
              (start (nth 1 fn))
              (end (nth 2 fn))
              (content (nth 3 fn)))
          (when (my/richfn--has-rich-content-p content)
            (my/richfn--log "Auto-converting footnote: %s" id)
            (let ((converted (funcall converter content)))
              (delete-region start end)
              (goto-char start)
              (insert (format "[fn:%s] %s\n" id converted)))))))))

(defvar my/richfn-debug nil "Enable debug messages for rich footnotes.")

(defun my/richfn--log (fmt &rest args)
  "Log message if debugging enabled."
  (when my/richfn-debug
    (apply #'message (concat "RichFN: " fmt) args)))

(defun my/richfn--has-rich-content-p (text)
  "Check if TEXT contains rich org content."
  (and text
       (string-match-p
        (rx (or
             (seq bol (+ "*") " ")
             (seq bol (* space) (any "-+") " ")
             (seq bol (* space) (+ digit) (any ".)") " ")
             (seq bol (* space) "|")
             (seq bol (* space) "#+begin_src")
             (seq bol (* space) "#+BEGIN_SRC")
             (seq "[fn:")))
        text)))

(defun my/richfn--get-footnote-definitions ()
  "Get all footnote definitions as alist of (id start end content)."
  (let ((footnotes nil))
    (save-excursion
      (goto-char (point-min))
      (while (re-search-forward "^\\[fn:\\([^]]+\\)\\][ \t]*" nil t)
        (let* ((id (match-string-no-properties 1))
               (start (match-beginning 0))
               (content-start (point))
               (end (save-excursion
                      (if (re-search-forward "^\\[fn:\\|^\\*+ \\|\\'" nil t)
                          (match-beginning 0)
                        (point-max))))
               (content (string-trim
                         (buffer-substring-no-properties content-start end))))
          (push (list id start end content) footnotes))))
    (nreverse footnotes)))

(defun my/richfn--parse-rich-blocks ()
  "Parse #+BEGIN_RICHFN blocks. Returns list of (id start end content)."
  (let ((blocks nil))
    (save-excursion
      (goto-char (point-min))
      (while (re-search-forward "^[ \t]*#\\+BEGIN_RICHFN[ \t]+\\([^ \t\n]+\\)" nil t)
        (let* ((id (match-string-no-properties 1))
               (block-start (line-beginning-position))
               (content-start (1+ (line-end-position)))
               (depth 1)
               (block-end nil)
               (content-end nil))
          ;; Find matching END_RICHFN (handle nesting)
          (while (and (> depth 0)
                      (re-search-forward "^[ \t]*#\\+\\(BEGIN\\|END\\)_RICHFN" nil t))
            (if (string= (match-string 1) "BEGIN")
                (setq depth (1+ depth))
              (setq depth (1- depth))
              (when (= depth 0)
                (setq content-end (line-beginning-position))
                (setq block-end (line-end-position)))))
          (when block-end
            (let ((content (string-trim
                            (buffer-substring-no-properties content-start content-end))))
              (push (list id block-start block-end content) blocks))))))
    (nreverse blocks)))

(defun my/richfn--org-to-latex (text)
  "Convert org TEXT to LaTeX for footnote."
  (with-temp-buffer
    (insert text)
    
    ;; Process nested RICHFN blocks first (recursive)
    (goto-char (point-min))
    (while (re-search-forward "^[ \t]*#\\+BEGIN_RICHFN[ \t]+\\([^ \t\n]+\\)" nil t)
      (let* ((nested-id (match-string-no-properties 1))
             (nested-start (line-beginning-position))
             (content-start (1+ (line-end-position)))
             (depth 1)
             (nested-end nil)
             (content-end nil))
        (while (and (> depth 0)
                    (re-search-forward "^[ \t]*#\\+\\(BEGIN\\|END\\)_RICHFN" nil t))
          (if (string= (match-string 1) "BEGIN")
              (setq depth (1+ depth))
            (setq depth (1- depth))
            (when (= depth 0)
              (setq content-end (line-beginning-position))
              (setq nested-end (1+ (line-end-position))))))
        (when nested-end
          (let* ((nested-content (buffer-substring-no-properties content-start content-end))
                 (converted (my/richfn--org-to-latex nested-content)))
            (delete-region nested-start nested-end)
            (goto-char nested-start)
            (insert (format "\\rfnmark{B}\\rfntext{B}{%s}" converted))))))
    
    ;; Convert headings (20 levels)
    (goto-char (point-min))
    (while (re-search-forward "^\\(\\*+\\) \\(.*\\)$" nil t)
      (let ((level (min 20 (length (match-string 1))))
            (title (match-string 2)))
        (replace-match (format "\\\\fnh{%d}{%s}" level title) t t)))
    
    ;; Convert description lists
    (goto-char (point-min))
    (let ((in-desc nil)
          (desc-regions nil))
      (while (re-search-forward "^\\([ \t]*\\)- \\([^:\n]+\\) :: \\(.*\\)$" nil t)
        (let ((indent (match-string 1))
              (term (match-string 2))
              (desc (match-string 3))
              (start (match-beginning 0)))
          (unless in-desc
            (setq in-desc start))
          (replace-match (format "%s\\\\item[%s] %s" indent term desc) t t)))
      ;; Wrap description items
      (goto-char (point-min))
      (while (re-search-forward "\\(\\\\item\\[[^]]*\\].*\n\\)+" nil t)
        (let ((items (match-string 0))
              (start (match-beginning 0))
              (end (match-end 0)))
          (delete-region start end)
          (goto-char start)
          (insert "\\begin{fndescription}\n" items "\\end{fndescription}\n"))))
    
    ;; Convert unordered lists
    (goto-char (point-min))
    (while (re-search-forward "^\\([ \t]*\\)[-+] \\(.*\\)$" nil t)
      (let ((indent (match-string 1))
            (content (match-string 2)))
        (replace-match (format "%s\\\\item %s" indent content) t t)))
    
    ;; Convert ordered lists
    (goto-char (point-min))
    (while (re-search-forward "^\\([ \t]*\\)[0-9]+[.)] \\(.*\\)$" nil t)
      (let ((indent (match-string 1))
            (content (match-string 2)))
        (replace-match (format "%s\\\\itemenum %s" indent content) t t)))
    
    ;; Wrap unordered items
    (goto-char (point-min))
    (while (re-search-forward "\\(^[ \t]*\\\\item [^\n]*\n\\)+" nil t)
      (let ((items (match-string 0))
            (start (match-beginning 0))
            (end (match-end 0)))
        (delete-region start end)
        (goto-char start)
        (insert "\\begin{fnitemize}\n" items "\\end{fnitemize}\n")))
    
    ;; Wrap ordered items
    (goto-char (point-min))
    (while (re-search-forward "\\(^[ \t]*\\\\itemenum [^\n]*\n\\)+" nil t)
      (let ((items (replace-regexp-in-string "\\\\itemenum" "\\\\item" (match-string 0)))
            (start (match-beginning 0))
            (end (match-end 0)))
        (delete-region start end)
        (goto-char start)
        (insert "\\begin{fnenumerate}\n" items "\\end{fnenumerate}\n")))
    
    ;; Convert tables
    (goto-char (point-min))
    (my/richfn--convert-org-tables-latex)
    
    ;; Convert code blocks
    (goto-char (point-min))
    (while (re-search-forward "^[ \t]*#\\+\\(begin_src\\|BEGIN_SRC\\)\\(?: \\([^ \t\n]+\\)\\)?[^\n]*\n\\(\\(?:.*\n\\)*?\\)[ \t]*#\\+\\(end_src\\|END_SRC\\)" nil t)
      (let ((lang (or (match-string 2) ""))
            (code (match-string 3)))
        (replace-match (format "\\\\begin{fncode}[%s]\n%s\\\\end{fncode}" lang code) t t)))
    
    ;; Convert inline code
    (goto-char (point-min))
    (while (re-search-forward "~\\([^~\n]+\\)~" nil t)
      (replace-match "\\\\fnc{\\1}" t nil))
    (goto-char (point-min))
    (while (re-search-forward "=\\([^=\n]+\\)=" nil t)
      (replace-match "\\\\fnc{\\1}" t nil))
    
    ;; Convert footnote references
    (goto-char (point-min))
    (while (re-search-forward "\\[fn:\\([^]]+\\)\\]" nil t)
      (replace-match "\\\\rfnmark{B}" t t))
    
    (buffer-string)))

(defun my/richfn--convert-org-tables-latex ()
  "Convert all org tables in buffer to LaTeX."
  (goto-char (point-min))
  (while (re-search-forward "^[ \t]*|" nil t)
    (let ((table-start (line-beginning-position))
          (rows nil)
          (col-count 0)
          (caption nil))
      ;; Check for caption above table
      (save-excursion
        (forward-line -1)
        (when (looking-at "^[ \t]*#\\+\\(CAPTION\\|caption\\):[ \t]*\\(.*\\)$")
          (setq caption (match-string 2))
          (setq table-start (line-beginning-position))))
      ;; Parse table rows
      (goto-char table-start)
      (when caption (forward-line 1))
      (while (looking-at "^[ \t]*|\\(.*\\)|[ \t]*$")
        (let* ((line (match-string 1))
               (cells (split-string line "|"))
               (is-sep (and cells (string-match-p "^[-+]+$" (string-trim (car cells))))))
          (unless is-sep
            (setq col-count (max col-count (length cells)))
            (push (mapcar #'string-trim cells) rows)))
        (forward-line 1))
      (let ((table-end (point)))
        (setq rows (nreverse rows))
        ;; Build LaTeX table
        (let ((latex
               (concat
                (if caption
                    (format "\\begin{fntab}{|%s|}[%s]\n"
                            (mapconcat (lambda (_) "l") (number-sequence 1 col-count) "|")
                            caption)
                  (format "\\begin{fntab}{|%s|}\n"
                          (mapconcat (lambda (_) "l") (number-sequence 1 col-count) "|")))
                "\\hline\n"
                (mapconcat
                 (lambda (row)
                   (concat (mapconcat #'identity row " & ") " \\\\"))
                 rows
                 "\n")
                "\n\\hline\n\\end{fntab}\n")))
          (delete-region table-start table-end)
          (goto-char table-start)
          (insert latex))))))

(defun my/richfn--org-to-typst (text)
  "Convert org TEXT to Typst for footnote."
  (with-temp-buffer
    (insert text)
    
    ;; Process nested RICHFN blocks
    (goto-char (point-min))
    (while (re-search-forward "^[ \t]*#\\+BEGIN_RICHFN[ \t]+\\([^ \t\n]+\\)" nil t)
      (let* ((nested-start (line-beginning-position))
             (content-start (1+ (line-end-position)))
             (depth 1)
             (nested-end nil)
             (content-end nil))
        (while (and (> depth 0)
                    (re-search-forward "^[ \t]*#\\+\\(BEGIN\\|END\\)_RICHFN" nil t))
          (if (string= (match-string 1) "BEGIN")
              (setq depth (1+ depth))
            (setq depth (1- depth))
            (when (= depth 0)
              (setq content-end (line-beginning-position))
              (setq nested-end (1+ (line-end-position))))))
        (when nested-end
          (let* ((nested-content (buffer-substring-no-properties content-start content-end))
                 (converted (my/richfn--org-to-typst nested-content)))
            (delete-region nested-start nested-end)
            (goto-char nested-start)
            (insert (format "#footnote[%s]" converted))))))
    
    ;; Convert headings
    (goto-char (point-min))
    (while (re-search-forward "^\\(\\*+\\) \\(.*\\)$" nil t)
      (let ((level (min 20 (length (match-string 1))))
            (title (match-string 2)))
        (replace-match (format "#fnh(%d)[%s]" level title) t t)))
    
    ;; Lists work natively in Typst
    
    ;; Convert tables
    (goto-char (point-min))
    (my/richfn--convert-org-tables-typst)
    
    ;; Convert code blocks
    (goto-char (point-min))
    (while (re-search-forward "^[ \t]*#\\+\\(begin_src\\|BEGIN_SRC\\)\\(?: \\([^ \t\n]+\\)\\)?[^\n]*\n\\(\\(?:.*\n\\)*?\\)[ \t]*#\\+\\(end_src\\|END_SRC\\)" nil t)
      (let ((lang (or (match-string 2) ""))
            (code (string-trim-right (match-string 3))))
        (if (string= lang "")
            (replace-match (format "#fn-code[```\n%s```]" code) t t)
          (replace-match (format "#fn-code(lang: \"%s\")[```%s\n%s```]" lang lang code) t t))))
    
    ;; Convert inline code
    (goto-char (point-min))
    (while (re-search-forward "~\\([^~\n]+\\)~" nil t)
      (replace-match "`\\1`" t nil))
    (goto-char (point-min))
    (while (re-search-forward "=\\([^=\n]+\\)=" nil t)
      (replace-match "`\\1`" t nil))
    
    ;; Convert footnote references
    (goto-char (point-min))
    (while (re-search-forward "\\[fn:\\([^]]+\\)\\]" nil t)
      (replace-match "#footnote[]" t t))
    
    (buffer-string)))

(defun my/richfn-preprocess-buffer (backend)
  "Preprocess rich footnotes before export to BACKEND."
  (my/richfn--log "Preprocessing for backend: %s" backend)
  
  (let ((converter (my/richfn--get-converter backend)))
    
    ;; Process explicit RICHFN blocks
    (let ((blocks (my/richfn--parse-rich-blocks)))
      (my/richfn--log "Found %d RICHFN blocks" (length blocks))
      (dolist (block (nreverse blocks))
        (let ((id (nth 0 block))
              (start (nth 1 block))
              (end (nth 2 block))
              (content (nth 3 block)))
          (my/richfn--log "Processing block: %s" id)
          (let ((converted (funcall converter content)))
            (delete-region start (1+ end))
            (save-excursion
              (goto-char (point-max))
              (insert (format "\n[fn:%s] %s\n" id converted)))))))
    
    ;; Auto-detect rich content in footnotes
    (let ((footnotes (my/richfn--get-footnote-definitions)))
      (my/richfn--log "Checking %d footnotes" (length footnotes))
      (dolist (fn (nreverse footnotes))
        (let ((id (nth 0 fn))
              (start (nth 1 fn))
              (end (nth 2 fn))
              (content (nth 3 fn)))
          (when (my/richfn--has-rich-content-p content)
            (my/richfn--log "Converting footnote: %s" id)
            (let ((converted (funcall converter content)))
              (delete-region start end)
              (goto-char start)
              (insert (format "[fn:%s] %s\n" id converted)))))))))

    #table(..args)
  ]

    fill: luma(245),
    inset: 5pt,
    radius: 2pt,
    width: 100%,

  )

#let fnc(body) = raw(body)
