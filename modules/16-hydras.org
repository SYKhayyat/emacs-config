(use-package hydra
:demand t
:config
(defhydra hydra-seforim (:color blue :hint nil)
"
+--------------------------------------------------------------------------+
| S E F O R I M 3 . 0 --- S I X S E A R C H M E T H O D S |
+--------------------------------------------------------------------------+
| FILE NAME SEARCH TEXT SEARCH (inside documents) |
| f: plocate (instant, exact) s: rg (notes: .org/.tex) |
| F: fd (live filesystem) S: rga (PDFs/EPUBs) |
| z: fzf (fuzzy, typo-tolerant) r: recoll (boolean, advanced) |
+--------------------------------------------------------------------------+
| READING & NAVIGATION LIBRARY MANAGEMENT |
| w: reading room toggle l: study log (recent files) |
| e: open EPUB (nov) ?: library status |
| d: jump to daf (PDF/Org) b: browse seforim directory |
| o: consult outline |
+--------------------------------------------------------------------------+
"
;; File name search
("f" seforim-find-name)
("F" seforim-find-live)
("z" seforim-find-fzf)
;; Text search inside documents (Updated to match consolidated 14-seforim)
("s" seforim-search-library)
("S" (lambda () (interactive) (seforim-search-library t)))
("r" seforim-search-recoll)
;; Reading & navigation
("w" my/toggle-reading-room)
("e" nov-open)
("d" seforim-goto-daf)
("o" consult-outline)
("b" (lambda () (interactive) (dired seforim-directory)))
;; Library management
("l" seforim-show-log)
("?" seforim-status)
("q" nil))

(global-set-key (kbd "C-c S") #'hydra-seforim/body)
