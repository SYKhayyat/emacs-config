  (defun seforim-orderless-hebrew-dispatch (pattern _index _total)
    "Multi-method Hebrew Matching.
     -term (exclude), =term (literal), term (flex + spelling variants)."
    (cond
     ((string-prefix-p "-" pattern) `(orderless-without-literal . ,(substring pattern 1)))
     ((string-prefix-p "=" pattern) `(orderless-literal . ,(substring pattern 1)))
     ((string-match-p "\\`[א-ת- ]+\\'" pattern)

        (setq regex (replace-regexp-in-string "ו" "ו?" regex))
        (setq regex (replace-regexp-in-string "י" "י?" regex))
        (mapconcat (lambda (c) (regexp-quote (char-to-string c))) regex ".*")))))

(defun my/kill-other-buffers ()
  "Kill all other buffers."
  (interactive)
  (let ((kept 0) (killed 0))
    (dolist (buf (buffer-list))
      (let ((name (buffer-name buf)))
        (cond
         ((eq buf (current-buffer)) (cl-incf kept))
         ((string-prefix-p " " name) (cl-incf kept))
         ((string-prefix-p "*" name) (cl-incf kept))
         (t (kill-buffer buf) (cl-incf killed)))))
    (message "Killed %d, kept %d" killed kept)))

;; ═══════════════════════════════════════════════════════════════════════════
;; NESTED FOOTNOTES FOR ORG
;; Syntax: [fn:1] for level 1, [fn:1:2] for level 2, [fn:1:2:3] for level 3
;; ═══════════════════════════════════════════════════════════════════════════

(defvar my/org-footnote-level-commands
  '((1 . "footnote")
    (2 . "footnoteB")
    (3 . "footnoteC")
    (4 . "footnoteD")
    (5 . "footnoteE")
    (6 . "footnoteF")
    (7 . "footnoteG")
    (8 . "footnoteH")
    (9 . "footnoteI")
    (10 . "footnoteJ"))
  "Mapping of footnote nesting levels to LaTeX commands.")

(defun my/org-nested-footnote-level (label)
  "Determine nesting level from footnote LABEL like 'fn:1:2:3'."
  (length (split-string (replace-regexp-in-string "^fn:" "" label) ":")))

(defun my/org-convert-nested-footnotes (backend)
  "Convert nested footnote syntax to bigfoot commands for LaTeX export."
  (when (memq backend '(latex beamer))
    (let ((fn-def-re "^\\[fn:\\([^]]+\\)\\]")
          (fn-ref-re "\\[fn:\\([^]]+\\)\\]"))
      ;; Process footnote definitions
      (save-excursion
        (goto-char (point-min))
        (while (re-search-forward fn-def-re nil t)
          (let* ((label (match-string 1))
                 (level (my/org-nested-footnote-level label))
                 (cmd (cdr (assoc level my/org-footnote-level-commands))))
            (when (and cmd (> level 1))
              ;; Mark this footnote for special handling
              (put-text-property (match-beginning 0) (match-end 0)
                                 'org-footnote-level level))))))))

(with-eval-after-load 'ox
  (add-hook 'org-export-before-processing-hook #'my/org-convert-nested-footnotes))

;; Custom footnote export
(with-eval-after-load 'ox-latex
  (defun my/org-latex-footnote-reference (footnote-reference _contents info)
    "Export footnote with proper nesting level."
    (let* ((label (org-element-property :label footnote-reference))
           (level (if label (my/org-nested-footnote-level label) 1))
           (cmd (or (cdr (assoc level my/org-footnote-level-commands)) "footnote"))
           (def (org-export-get-footnote-definition footnote-reference info)))
      (format "\\%s{%s}" cmd
              (org-trim (org-export-data def info)))))

  ;; Advice to use our custom exporter for nested footnotes
  (advice-add 'org-latex-footnote-reference :override #'my/org-latex-footnote-reference))

  "Fix all footnotes for RTL."

          (insert ?\x200F)

(defun seforim--open-file (file)
  "Open FILE. C-u or Shift+Enter opens in new tab."
  (let ((tab-wanted (or current-prefix-arg (memq 'shift (event-modifiers last-command-event)))))
    (if tab-wanted
        (let ((tab-name (file-name-base file)))
          (tab-bar-new-tab)
          (find-file file)
          (tab-bar-rename-tab tab-name))
      (find-file file))
    (seforim--log-session file)))

(defun seforim-find ()
  "Dynamic filename search via plocate."
  (interactive)
  (let ((file (consult--read
               (consult--async-process
                (lambda (input) (list "plocate" "-i" "-d" "/var/cache/locatedb" "-r" (concat (expand-file-name seforim-directory) ".*" input))))
               :prompt "Find Sefer: "
               :category 'seforim-file)))
    (when file (seforim--open-file file))))

(defun seforim-search ()
  "Dynamic content search via Recoll."
  (interactive)
  (let ((cand (consult--read
               (consult--async-process
                (lambda (input) (list "recoll" "-t" "-q" input "dir:" (expand-file-name seforim-directory))))
               :prompt "Search Content: "
               :category 'recoll-result)))
    (when (and cand (string-match "\\(/[^ ]+\\)" cand))
      (seforim--open-file (match-string 1 cand)))))

(defun seforim-find-live ()
  "Live filesystem search via fzf."
  (interactive)
  (let ((default-directory seforim-directory))
    (consult-find default-directory)))

(defun seforim-search-live ()
  "Live content search via rga."
  (interactive)
  (let ((consult-ripgrep-args "rga --null --column --line-number --no-heading --color=never --smart-case --max-columns=4096 --max-columns-preview"))
    (consult-ripgrep seforim-directory)))
#+end_src

(defun seforim-embark-open-tab (file)
  "Open file in new tab without closing minibuffer."
  (interactive "fSefer: ")
  (let ((tab-name (file-name-base file)))
    (tab-bar-new-tab)
    (find-file file)
    (tab-bar-rename-tab tab-name)))

(defun seforim-embark-copy-citation (file)
  "Copy file name as citation."
  (interactive "fSefer: ")
  (kill-new (file-name-nondirectory file))
  (message "Citation copied: %s" (file-name-nondirectory file)))

(defvar-keymap seforim-embark-file-map
  :doc "Scholarly actions for Seforim library files."
  "t" #'seforim-embark-open-tab
  "c" #'seforim-embark-copy-citation
  "s" (lambda (file) (interactive) (let ((default-directory (file-name-directory file))) (consult-line))))

(defun seforim-status ()
  (interactive)
  (let ((plocate (file-exists-p "/var/cache/locatedb")) (recoll (file-directory-p (expand-file-name "~/.recoll"))))
    (message "Library: OK | plocate: %s | Recoll: %s" (if plocate "OK" "MISS") (if recoll "OK" "MISS"))))

(defun seforim--num-to-heb (n)
  (let ((ones '("" "א" "ב" "ג" "ד" "ה" "ו" "ז" "ח" "ט"))
        (tens '("" "י" "כ" "ל" "מ" "נ" "ס" "ע" "פ" "צ"))
        (h (/ n 100)) (t-val (/ (% n 100) 10)) (o (% n 10)))
    (concat (nth h '("" "ק" "ר" "ש" "ת"))
            (cond ((= (% n 100) 15) "טו") ((= (% n 100) 16) "טז") (t (concat (nth t-val tens) (nth o ones)))))))

(defun seforim-goto-daf ()
  (interactive)
  (let* ((daf (read-number "Daf: " 2)) (amud (completing-read "Amud: " '("א" "ב") nil t)))
    (cond
     ((derived-mode-p 'pdf-view-mode) (my/pdf-jump-to-daf daf amud))
     (t (let ((target (format "דף %s ע\"%s" (seforim--num-to-heb daf) amud)))
          (goto-char (point-min))
          (if (re-search-forward (regexp-quote target) nil t) (progn (org-reveal) (recenter)) (message "Daf not found.")))))))

(defun seforim--log-session (file)
  (when (string-prefix-p (expand-file-name seforim-directory) (expand-file-name file))
    (let ((entry (cons (file-name-nondirectory file) (current-time))))
      (setq seforim--log-list (cons entry (delete entry seforim--log-list)))
      (make-directory (file-name-directory seforim-study-log-path) t)
      (with-temp-file seforim-study-log-path (insert (prin1-to-string seforim--log-list))))))

(defun seforim-show-log ()
  (interactive)
  (when (file-exists-p seforim-study-log-path) (with-temp-buffer (insert-file-contents seforim-study-log-path) (setq seforim--log-list (read (current-buffer)))))
  (let* ((candidates (mapcar (lambda (x) (format "%s (%s)" (car x) (format-time-string "%H:%M" (cdr x)))) seforim--log-list))
         (choice (completing-read "History: " candidates nil t)))
    (when choice (let* ((fname (car (split-string choice " ("))) (matches (directory-files-recursively seforim-directory (regexp-quote fname)))) (when matches (seforim--open-file (car matches)))))))
