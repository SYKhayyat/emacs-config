;;; 15-hydras.el --- Hydra definitions -*- lexical-binding: t; -*-

(use-package hydra
  :demand t
  :config

  ;; Direction and spell hydra
  (defhydra hydra-direction (:color blue :hint nil)
    "
Direction: _r_: RTL  _l_: LTR  _t_: toggle  _B_: bidi
Spell:     _h_: Hebrew  _e_: English  _b_: both
Writing:   _w_: focused  _W_: width
"
    ("r" my/set-rtl)
    ("l" my/set-ltr)
    ("t" my/toggle-direction)
    ("B" my/toggle-bidi-reordering)
    ("h" my/spell-hebrew)
    ("e" my/spell-english)
    ("b" my/spell-both)
    ("w" my/toggle-focused-writing)
    ("W" my/set-focused-width)
    ("q" nil))

  (global-set-key (kbd "C-c D") #'hydra-direction/body)

  ;; Org-roam hydra
  (defhydra hydra-roam (:color blue :hint nil)
    "
Roam: _f_: find  _i_: insert  _c_: capture  _l_: buffer
      _d_: daily  _D_: date  _g_: graph  _u_: UI
      _h_: hebrew  _R_: research  _t_: tag
"
    ("f" org-roam-node-find)
    ("i" org-roam-node-insert)
    ("c" org-roam-capture)
    ("l" org-roam-buffer-toggle)
    ("d" org-roam-dailies-goto-today)
    ("D" org-roam-dailies-goto-date)
    ("g" org-roam-graph)
    ("u" org-roam-ui-mode)
    ("h" my/org-roam-find-hebrew)
    ("R" my/org-roam-find-research)
    ("t" org-roam-tag-add)
    ("q" nil))

  (global-set-key (kbd "C-c N") #'hydra-roam/body)

  ;; Seforim hydra
  (defhydra hydra-seforim (:color blue :hint nil)
    "
Seforim: _f_: find  _r_: recent  _s_: search  _b_: browse
         _o_: outline  _d_: daf  _D_: daf (current)
         _I_: reindex  _?_: status
"
    ("f" seforim-find)
    ("r" seforim-find-recent)
    ("s" seforim-search)
    ("b" seforim-browse)
    ("o" seforim-outline)
    ("d" seforim-goto-daf)
    ("D" seforim-goto-daf-current)
    ("I" seforim-reindex)
    ("?" seforim-status)
    ("q" nil))

  (global-set-key (kbd "C-c S") #'hydra-seforim/body)

  ;; Rich footnotes hydra
  (defhydra hydra-rich-footnotes (:color blue :hint nil)
    "
Rich Footnotes
Insert:   _l_: LaTeX    _t_: Typst      _c_: ConTeXt
Preamble: _p_: LaTeX    _T_: Typst      _C_: ConTeXt
"
    ("l" my/insert-rich-footnote-latex)
    ("t" my/insert-rich-footnote-typst)
    ("c" my/insert-rich-footnote-context)
    ("p" my/insert-latex-rich-footnotes-preamble)
    ("T" my/insert-typst-rich-footnotes-preamble)
    ("C" my/insert-context-rich-footnotes-preamble)
    ("q" nil))

  (global-set-key (kbd "C-c F") #'hydra-rich-footnotes/body)

  ;; Document hydra
  (defhydra hydra-document (:color blue :hint nil)
    "
Documents
Preambles: _p_: LaTeX Hebrew  _t_: Typst Hebrew  _c_: ConTeXt Hebrew
Rich FN:   _l_: LaTeX Rich    _T_: Typst Rich    _C_: ConTeXt Rich
Create:    _n_: New document
"
    ("p" my/insert-latex-hebrew-preamble)
    ("t" my/insert-typst-hebrew-preamble)
    ("c" my/insert-context-hebrew-preamble)
    ("l" my/insert-latex-rich-footnotes-preamble)
    ("T" my/insert-typst-rich-footnotes-preamble)
    ("C" my/insert-context-rich-footnotes-preamble)
    ("n" my/create-hebrew-document)
    ("q" nil))

  (global-set-key (kbd "C-c P") #'hydra-document/body))

(provide '15-hydras)
