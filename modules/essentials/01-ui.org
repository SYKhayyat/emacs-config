;;; 01-ui.el --- User interface settings -*- lexical-binding: t; -*-

;; Startup
(setq inhibit-startup-message t
      initial-scratch-message nil
      ring-bell-function 'ignore)

;; Remove UI bars
(scroll-bar-mode -1)
(tool-bar-mode -1)
(tooltip-mode -1)
(menu-bar-mode -1)
(set-fringe-mode 10)

;; Line numbers
(setq display-line-numbers-type 'relative
      display-line-numbers-width 3)

(add-hook 'prog-mode-hook #'display-line-numbers-mode)
(add-hook 'text-mode-hook #'display-line-numbers-mode)
(add-hook 'conf-mode-hook #'display-line-numbers-mode)

(dolist (mode '(org-mode-hook term-mode-hook shell-mode-hook
                eshell-mode-hook vterm-mode-hook pdf-view-mode-hook))
  (add-hook mode (lambda () (display-line-numbers-mode 0))))

;; Tall / large buffers degrade gracefully.
;;
;; Relative numbering is the worst case on a tall file: Emacs recomputes the
;; offset of every visible line on each scroll.  For a truly large file,
;; `my/apply-large-file-settings' (00-core) has already put this buffer in
;; `my/large-file-mode' and switched line numbers off, so this hook stands
;; down there -- it only touches merely-tall buffers, where it falls back to
;; absolute numbering.
(defun my/ui-large-or-tall-p ()
  "Non-nil when the current file is too large or tall for relative numbers."
  (and buffer-file-name
       (not (bound-and-true-p my/large-file-mode))
       (> (line-number-at-pos (point-max)) my/medium-file-lines)))

(defun my/ui-degrade-line-numbers ()
  "Use absolute (not relative) line numbers in tall buffers."
  (when (my/ui-large-or-tall-p)
    (setq display-line-numbers t)))

(add-hook 'find-file-hook #'my/ui-degrade-line-numbers)

;; Visual aids
(global-hl-line-mode 1)
(show-paren-mode 1)
(setq show-paren-delay 0)
(column-number-mode 1)
(fset 'yes-or-no-p 'y-or-n-p)
(save-place-mode 1)
(global-auto-revert-mode 1)
(setq global-auto-revert-non-file-buffers t)

;; Scrolling
(setq scroll-conservatively 101
      scroll-margin 3
      scroll-preserve-screen-position t)

;; Soft line wrapping at word boundaries.
;;
;; `visual-line-mode' sets `word-wrap' locally, but a plain text/fundamental
;; buffer (and `so-long' on long-line buffers) wraps at character level,
;; splitting a Hebrew or otherwise long word across two lines. Defaulting
;; `word-wrap' to t makes every soft break land at a word boundary, so the
;; letters of a word stay together no matter which minor mode is active.
(setq-default word-wrap t)

;; Frame
(add-to-list 'default-frame-alist '(height . 35))
;; Files and backups
(setq make-backup-files nil
      auto-save-default nil
      create-lockfiles nil)

;; Encoding
(set-default-coding-systems 'utf-8)
(set-terminal-coding-system 'utf-8)
(set-keyboard-coding-system 'utf-8)
(prefer-coding-system 'utf-8)

;; Recent files
(use-package recentf
  :ensure nil
  :config
  (setq recentf-max-saved-items 200
        recentf-max-menu-items 15
        recentf-auto-cleanup 'never)
  (add-to-list 'recentf-exclude "\\.?cache")
  (add-to-list 'recentf-exclude ".*\\.elc$")
  (add-to-list 'recentf-exclude "/nix/store/")
  (recentf-mode 1))

;; Save history
(use-package savehist
  :ensure nil
  :config
  (setq history-length 1000
        savehist-additional-variables
        '(search-ring regexp-search-ring kill-ring compile-command))
  (savehist-mode 1))

(use-package nerd-icons :demand t)

  :demand t

        pulsar-face 'pulsar-magenta)

  (add-hook 'next-error-hook #'pulsar-pulse-line))

;; Fonts
;; Script-specific fontsets belong to whoever needs that script: `my/pick-font'
;; is exported so an extras module can pick a face for its own charset without
;; this file having to know which scripts you read.
(defun my/pick-font (families)
  "Return the first family in FAMILIES that is actually installed, or nil."
  (seq-find (lambda (f) (member f (font-family-list))) families))

(when (display-graphic-p)
  ;; Pick the first AVAILABLE family so a missing font never errors and every
  ;; OS gets a sane face (NixOS/Linux, Windows, macOS).
  (let ((mono (my/pick-font '("JetBrains Mono" "Cascadia Code" "Iosevka"
                              "DejaVu Sans Mono" "Consolas" "Menlo" "Courier New"))))
    (when mono
      (set-face-attribute 'default nil :family mono :height 90 :weight 'regular)))
  (when (fboundp 'set-fontset-font)
    (set-fontset-font t 'unicode (font-spec :family "DejaVu Sans") nil 'append))
  (set-face-attribute 'minibuffer-prompt nil :height 80)
  (set-face-attribute 'mode-line nil :height 80)
  (set-face-attribute 'mode-line-inactive nil :height 80))
