;; Tall / large buffers degrade gracefully.
;;
;; Relative numbering is the worst case on a tall file: Emacs recomputes the
;; offset of every visible line on each scroll.  For a truly large file,
;; `my/apply-large-file-settings' (00-core) has already put this buffer in
;; `my/large-file-mode' and switched line numbers off, so this hook stands
;; down there -- it only touches merely-tall buffers, where it falls back to
;; absolute numbering.
(defun my/ui-large-or-tall-p ()
  "Non-nil when the current file is too large or tall for relative numbers."
  (and buffer-file-name
       (not (bound-and-true-p my/large-file-mode))
       (> (line-number-at-pos (point-max)) my/medium-file-lines)))

(defun my/ui-degrade-line-numbers ()
  "Use absolute (not relative) line numbers in tall buffers."
  (when (my/ui-large-or-tall-p)
    (setq display-line-numbers t)))

(add-hook 'find-file-hook #'my/ui-degrade-line-numbers)

;; Soft line wrapping at word boundaries.
;;
;; `visual-line-mode' sets `word-wrap' locally, but a plain text/fundamental
;; buffer (and `so-long' on long-line buffers) wraps at character level,
;; splitting a Hebrew or otherwise long word across two lines. Defaulting
;; `word-wrap' to t makes every soft break land at a word boundary, so the
;; letters of a word stay together no matter which minor mode is active.
(setq-default word-wrap t)

;; Files and backups
;; Emacs defaults are t/t/t for all three.  This config is a daily driver, and
;; the trade is backwards: auto-save's idle timer costs a few cheap wakeups a
;; minute, while having it off means a crash, a restart, or an OOM kill loses
;; every unsaved buffer.  A SIGSEGV in the font-rendering path is already on the
;; record (#30), and systemd restarts the daemon from it.
;;
;; Lockfiles are off by the same stroke, and combined with init.el's
;; `vc-handled-backends '(Git)' -- one supported VCS -- two Emacs instances can
;; clobber each other silently (#47).
;;
;; Backups stay off: they are the least costly of the three to lose, and they
;; are the only one that writes on every save.
(setq make-backup-files nil
      auto-save-default t
      auto-save-time (* 10 60)
      create-lockfiles t)

(defun my/drop-none-style (box)
  "Return BOX with the (:style none) pair removed, rest preserved."
  (let ((kept nil))
    (while box
      (let ((key (pop box))
            (val (pop box)))
        (unless (and (eq key :style) (eq val 'none))
          (push key kept)
          (push val kept))))
    (nreverse kept)))

(defun my/sanitize-spec-tree (tree)
  "Return TREE with every :box carrying (:style none) repaired.
Emacs 30 rejects :style none as an invalid face box; doom-themes 2.3.0
sets it on the custom-button faces.  Overriding the faces afterwards
cannot help, because face realization validates every stored spec on
the way to the winner, so the invalid one is rewritten here, at install
time, the same way upstream fixed it: the style pair is dropped and the
rest of the box is kept."
  (cond ((not (consp tree)) tree)
        ((and (eq (car tree) :box)
              (consp (cdr tree))
              (listp (cadr tree))
              (eq (plist-get (cadr tree) :style) 'none))
         (cons :box (cons (my/drop-none-style (cadr tree))
                          (my/sanitize-spec-tree (cddr tree)))))
        (t (cons (my/sanitize-spec-tree (car tree))
                 (my/sanitize-spec-tree (cdr tree))))))

(defun my/sanitize-face-spec (spec)
  "Run `my/sanitize-spec-tree' over one theme face SPEC."
  (my/sanitize-spec-tree spec))

;; ── `custom-theme-set-faces' was the only door, and there are four ────────
;;
;; The sanitizer above was written against the one entry point doom-themes
;; happens to use -- `custom-theme-set-faces' -- and that turned out to be the
;; least likely of the four to be the one that bites:
;;
;;   custom-theme-set-faces  stores the spec.  Emacs validates it later, on the
;;                           way through redisplay, so the error is raised in
;;                           whatever Lisp was running at that moment.  The
;;                           report that prompted this blamed 11-pdf and
;;                           16-seforim-integration; neither file mentions a
;;                           face, and neither is at fault.
;;   custom-set-faces        same shape, same deferred failure.
;;   face-spec-set           validates *eagerly*.  This is the one that threw.
;;   set-face-attribute      validates eagerly too, and takes a plist rather
;;                           than a spec tree, so it needs its own walker.
;;
;; All four are advised, because the cost of missing one is a session that
;; starts with two modules "failed to load" for a reason neither of them
;; caused.  The two spec-tree entry points share an advice; the other two take
;; arguments of a different shape and get their own.

(defun my/sanitize-box (box)
  "Return BOX with a `(:style none)' pair dropped; BOX itself if it is clean."
  (if (and (consp box) (eq (plist-get box :style) 'none))
      (my/drop-none-style box)
    box))

(defun my/sanitize-args-boxes (args)
  "Return plist ARGS with every `:box' value passed through `my/sanitize-box'."
  (let ((out nil))
    (while args
      (let ((key (pop args)))
        (if (null args)
            (push key out)              ; odd-length plist: keep the bare key
          (let ((val (pop args)))
            (push key out)
            (push (if (eq key :box) (my/sanitize-box val) val) out)))))
    (nreverse out)))

(defun my/sanitize-theme-faces-advice (orig theme &rest specs)
  "Around advice for `custom-theme-set-faces' running `my/sanitize-face-spec'."
  (apply orig theme (mapcar #'my/sanitize-face-spec specs)))

(defun my/sanitize-custom-faces-advice (orig &rest specs)
  "Around advice for `custom-set-faces', which takes specs and no theme."
  (apply orig (mapcar #'my/sanitize-face-spec specs)))

(defun my/sanitize-face-spec-set-advice (orig face spec &rest args)
  "Around advice for `face-spec-set', whose SPEC is a spec tree."
  (apply orig face (my/sanitize-face-spec spec) args))

(defun my/sanitize-set-face-attribute-advice (orig face frame &rest args)
  "Around advice for `set-face-attribute', whose ARGS are a plist."
  (apply orig face frame (my/sanitize-args-boxes args)))

(advice-add 'custom-theme-set-faces :around #'my/sanitize-theme-faces-advice)
(advice-add 'custom-set-faces :around #'my/sanitize-custom-faces-advice)
(advice-add 'face-spec-set :around #'my/sanitize-face-spec-set-advice)
(advice-add 'set-face-attribute :around #'my/sanitize-set-face-attribute-advice)

;; Fonts
;; Script-specific fontsets belong to whoever needs that script: `my/pick-font'
;; is exported so an extras module can pick a face for its own charset without
;; this file having to know which scripts you read.
(defun my/pick-font (families)
  "Return the first family in FAMILIES that is actually installed, or nil."
  (seq-find (lambda (f) (member f (font-family-list))) families))

(when (display-graphic-p)
  ;; Pick the first AVAILABLE family so a missing font never errors and every
  ;; OS gets a sane face (NixOS/Linux, Windows, macOS).
  (let ((mono (my/pick-font '("JetBrains Mono" "Cascadia Code" "Iosevka"
                              "DejaVu Sans Mono" "Consolas" "Menlo" "Courier New"))))
    (when mono
      (set-face-attribute 'default nil :family mono :height 90 :weight 'regular)))
  (when (fboundp 'set-fontset-font)
    (set-fontset-font t 'unicode (font-spec :family "DejaVu Sans") nil 'append))
  (set-face-attribute 'minibuffer-prompt nil :height 80)
  (set-face-attribute 'mode-line nil :height 80)
  (set-face-attribute 'mode-line-inactive nil :height 80))
