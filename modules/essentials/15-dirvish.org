* Dirvish Configuration

;;; 15-dirvish.el --- Modern Dired replacement -*- lexical-binding: t; -*-

(use-package dirvish
  :demand t
  :init
  (require 'dirvish)
  ;; Replace standard dired with dirvish
  (dirvish-override-dired-mode)
  :custom
  ;; Quick-access paths.  An extras module appends its own with
  ;; `customize-set-variable' rather than editing this -- it has to go through
  ;; the defcustom's `:set', which is what regenerates the transient menu.
  (dirvish-quick-access-entries
   '(("h" "~/"                          "Home")
     ("d" "~/Downloads"                 "Downloads")
     ("n" "~/Documents/roam"            "Notes")
     ("o" "~/Documents/org"             "Org")))
  :config
  ;; Dirs before files in listings (`C-x d'), alphabetical within each group.
  ;; `--group-directories-first' is done by ls itself, and dirvish obeys this
  ;; variable unless `dirvish-listing-switches' is set.
  (setq dired-listing-switches "-al --group-directories-first")
  ;; Visual features
  (setq dirvish-attributes
        '(nerd-icons file-time file-size collapse git-msg))
  
  ;; Mode line setup
  (setq dirvish-mode-line-format
        '(:left (sort symlink) :right (omit yank index)))

  ;; IDE-style file-tree sidebar (<f8>): a clean, following project tree.
  ;; `require' the sub-feature so the `dirvish-side' command and its follow
  ;; mode are defined even when package autoloads were not generated (e.g. a
  ;; bare load-path setup) — this is what makes <f8> reliable everywhere.
  (require 'dirvish-side nil t)
  (setq dirvish-side-width 32)
  ;; Same reason, and this one is load-bearing: `dirvish-quick-access-entries'
  ;; is a defcustom in the `dirvish-quick-access' sub-feature, which `dirvish'
  ;; does not pull in -- it is autoloaded on the first `a'.  Until then the
  ;; variable is *void*, the `:custom' value above sits unapplied as a deferred
  ;; theme setting, and anything that appends to the list (extras/16) fails.
  ;; Loading it here defines the variable and applies that value at startup.
  (require 'dirvish-quick-access nil t)
  ;; `dirvish-side-follow-mode' is a global minor mode, so it must be *called*,
  ;; not `setq'd, to actually track the current file in the sidebar.
  (when (fboundp 'dirvish-side-follow-mode) (dirvish-side-follow-mode 1))

  :bind
  (("C-x d" . dirvish)
   ("<f8>"  . dirvish-side)     ; IDE-style file-tree sidebar toggle
   :map dirvish-mode-map
   ("a"   . dirvish-quick-access)
   ("f"   . dirvish-fd)
   ("TAB" . dirvish-subtree-toggle)
   ("M-t" . dirvish-layout-toggle)
   ("h"   . dired-up-directory)
   ("l"   . dired-find-file)))

(provide '15-dirvish)
#+end_src
