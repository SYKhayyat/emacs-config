;;; 12-programming.el --- Programming modes and LSP -*- lexical-binding: t; -*-

;; Eglot - LSP client
;; No `java-mode' here: `eglot-java-mode' already calls `eglot-ensure' via
;; `eglot-java--ensure', so the two hooks were both redundant and conflicting --
;; and `run-hooks' aborts on the first error, so a failed eglot-ensure meant
;; eglot-java-mode never ran at all (#31).
(use-package eglot
  :ensure nil
  :hook ((python-mode . eglot-ensure)
         (rust-mode . eglot-ensure)
         (nix-mode . eglot-ensure))
  :config
  (setq eglot-autoshutdown t
        eglot-send-changes-idle-time 0.5
        eglot-events-buffer-size 0)
  (with-eval-after-load 'typst-ts-mode
    (add-to-list 'eglot-server-programs '(typst-ts-mode . ("tinymist")))
    (add-hook 'typst-ts-mode-hook #'eglot-ensure)))

;; Python
(use-package python
  :ensure nil
  :mode ("\\.py\\'" . python-mode)
  :config
  (setq python-indent-offset 4))

;; Rust
(use-package rust-mode
  :mode "\\.rs\\'"
  :config
  (setq rust-format-on-save t))

(use-package cargo
  :hook (rust-mode . cargo-minor-mode))

;; Java
;; Gate on the server binary.  Without this, `eglot-java-mode' runs
;; `eglot-java--ensure' inside the mode hook, which does a SYNCHRONOUS
;; `url-retrieve-synchronously' of ~100MB of JDT and then:
;;
;;     (let ((b (find-file dest-abspath)))
;;       (switch-to-buffer b)
;;       (tar-untar-buffer)
;;       (kill-buffer b))
;;
;; i.e. your window switches to a tar buffer mid-find-file.  Gate on the tool, not
;; the OS -- the same axis `my/module-enabled-p' uses in init.el.
(use-package eglot-java
  :if (executable-find "jdtls")
  :hook (java-mode . eglot-java-mode)
  :config
  ;; Stop it rewriting eglot-server-programs behind your back.
  (setq eglot-java-eglot-server-programs-manual-updates t))

;; Nix
(use-package nix-mode
  :mode "\\.nix\\'"
  :config
  (with-eval-after-load 'eglot
    (add-to-list 'eglot-server-programs '(nix-mode . ("nil")))))

;; Markdown now lives in its own module (28-markdown), together with the
;; live side-by-side preview.  Keeping two `use-package markdown-mode' forms
;; in different modules meant both ran and the later one silently won.

;; Snippets
(use-package yasnippet
  :diminish yas-minor-mode
  :hook ((prog-mode . yas-minor-mode)
         (text-mode . yas-minor-mode))
  :config
  (yas-reload-all))

(use-package yasnippet-snippets
  :after yasnippet)

;; EditorConfig
(use-package editorconfig
  :diminish
  :demand t
  :config
  (editorconfig-mode 1))

;; Environment variables
;; envrc shells out to `direnv' on every buffer switch; without the binary it
;; is dead weight that errors in the echo area, so gate on the executable.
(use-package envrc
  :if (executable-find "direnv")
  :diminish
  :demand t
  :config
  (envrc-global-mode 1))

(provide '12-programming)
