;;; 00-hebrew.el --- Hebrew and RTL support -*- lexical-binding: t; -*-

(setq-default bidi-display-reordering t)
(setq-default bidi-paragraph-direction nil)

;; `bidi-inhibit-bpa' is deliberately not set here.  essentials/00-core sets the
;; global, and `my/apply-direction' below overrides it per buffer; a second
;; global assignment in the module that wants the BPA *on* for prose was only
;; ever confusing.

(defvar-local my/bidi-direction 'right-to-left)

;; Direction is three variables, not one, and only `bidi-paragraph-direction'
;; is per-buffer by default.  `visual-order-cursor-movement' and
;; `bidi-inhibit-bpa' are ordinary globals, so they need `setq-local' to stay
;; scoped to the buffer being set up.
;;
;; `visual-order-cursor-movement' makes C-f, C-b and the arrow keys follow what
;; is on screen rather than the order of the underlying characters.  In
;; monolingual Hebrew the two agree and it changes nothing; on a line mixing
;; scripts, the logical default lands the cursor somewhere the eye did not
;; follow it to.
;;
;; `bidi-inhibit-bpa' is set globally in essentials/00-core, where it is a
;; redisplay speedup and the right default for code.  It also switches off the
;; Bidirectional Parentheses Algorithm, which is what resolves a bracket's
;; direction from its pair rather than from the surrounding run -- so prose
;; wants it back on.  Clearing it per-buffer leaves the global default alone,
;; and a config that never loads this module is unaffected.

(defun my/apply-direction (direction)
  "Apply DIRECTION -- `right-to-left' or `left-to-right' -- to this buffer."
  (let ((rtl (eq direction 'right-to-left)))
    (setq-local bidi-paragraph-direction direction
                visual-order-cursor-movement rtl
                bidi-inhibit-bpa (not rtl))
    (setq my/bidi-direction direction)))

(defun my/setup-rtl-mode ()
  "Set up RTL for prose."
  (my/apply-direction 'right-to-left))

(defun my/setup-ltr-mode ()
  "Set up LTR for code."
  (my/apply-direction 'left-to-right))

(defun my/set-rtl ()
  "Set buffer to RTL."
  (interactive)
  (my/setup-rtl-mode)
  (message "Direction: RTL"))

(defun my/set-ltr ()
  "Set buffer to LTR."
  (interactive)
  (my/setup-ltr-mode)
  (message "Direction: LTR"))

(defun my/toggle-direction ()
  "Toggle RTL/LTR."
  (interactive)
  (if (eq bidi-paragraph-direction 'right-to-left)
      (my/set-ltr)
    (my/set-rtl)))

(defun my/toggle-bidi-reordering ()
  "Toggle bidi reordering."
  (interactive)
  (setq bidi-display-reordering (not bidi-display-reordering))
  (message "Bidi: %s" (if bidi-display-reordering "ON" "OFF")))

;; RTL for prose, LTR for code.  The typesetting modes are listed here rather
;; than in essentials/07-latex, 08-typst and 09-context: those are meant to work
;; as a plain LaTeX/Typst/ConTeXt setup for someone who never writes Hebrew, so
;; the direction hook belongs to the module that knows about direction.  Hooks
;; can be added to a mode that has not loaded yet, so ordering does not matter.
(dolist (hook '(org-mode-hook text-mode-hook markdown-mode-hook
                LaTeX-mode-hook typst-ts-mode-hook ConTeXt-mode-hook))
  (add-hook hook #'my/setup-rtl-mode))

;; nov (EPUB) sets its own major mode; RTL there is what makes Hebrew ebooks
;; readable.
(add-hook 'nov-mode-hook #'my/set-rtl)

(add-hook 'prog-mode-hook #'my/setup-ltr-mode)

;; ── Hebrew font ────────────────────────────────────────────────────────────
;; essentials/01-ui picks the monospace face and exports `my/pick-font'; the
;; Hebrew fontset is chosen here so a config without this module never carries
;; a list of Hebrew families it has no use for.
(when (and (display-graphic-p) (fboundp 'set-fontset-font))
  (let ((heb (my/pick-font '("David CLM" "Taamey Frank CLM" "Frank Ruehl CLM"
                             "SBL Hebrew" "Noto Sans Hebrew" "Arial Hebrew" "David"))))
    (when heb
      (set-fontset-font t 'hebrew (font-spec :family heb)))))

;; Emacs ships eight Hebrew and Yiddish Quail packages.  `hebrew' is the
;; standard Israeli layout and cannot enter niqqud or cantillation marks; the
;; two biblical methods exist for that.  F9 toggles the chosen one, C-u F9
;; picks a different one and makes it what F9 toggles from then on.

(defvar my/hebrew-input-methods
  '("hebrew" "hebrew-new" "hebrew-full" "hebrew-lyx"
    "hebrew-biblical-tiro" "hebrew-biblical-sil"
    "yiddish-royal" "yiddish-keyman")
  "Input methods offered by `my/select-hebrew-input'.")

(defvar my/hebrew-input-method "hebrew"
  "Input method `my/toggle-hebrew-input' activates.")

(defun my/select-hebrew-input (method)
  "Activate METHOD and make it the one `my/toggle-hebrew-input' activates."
  (interactive (list (completing-read "Hebrew input method: "
                                      my/hebrew-input-methods nil t)))
  (setq my/hebrew-input-method method)
  (set-input-method method))

(defun my/toggle-hebrew-input (&optional select)
  "Toggle `my/hebrew-input-method'.  With prefix SELECT, choose one first."
  (interactive "P")
  (cond (select (call-interactively #'my/select-hebrew-input))
        (current-input-method (deactivate-input-method))
        (t (set-input-method my/hebrew-input-method))))

(defun my/large-file-hook ()
  "Optimize large files."
  (when (> (buffer-size) (* 5 1024 1024))
    (setq-local bidi-display-reordering nil)
    (font-lock-mode -1)
    (message "Large file mode")))

(add-hook 'find-file-hook #'my/large-file-hook)

;; Keybindings
(global-set-key (kbd "<f9>") #'my/toggle-hebrew-input)
(global-set-key (kbd "C-c h i") #'my/select-hebrew-input)
(global-set-key (kbd "C-c d") #'my/toggle-direction)
(global-set-key (kbd "C-c r") #'my/set-rtl)
(global-set-key (kbd "C-c l") #'my/set-ltr)
(global-set-key (kbd "C-c B") #'my/toggle-bidi-reordering)

;; Hebrew-specific editing
(defun my/insert-hebrew-quotes ()
  "Insert Hebrew quotes (״)."
  (interactive)
  (insert "״")
  (save-excursion (insert "״")))

(defun my/insert-hebrew-parens ()
  "Insert Hebrew parentheses ()( in reverse order)."
  (interactive)
  (insert ")")
  (save-excursion (insert "(")))

(defun my/count-hebrew-words ()
  "Count Hebrew words in buffer."
  (interactive)
  (let ((count 0))
    (save-excursion
      (goto-char (point-min))
      (while (re-search-forward "[א-ת]+" nil t)
        (cl-incf count)))
    (message "Hebrew words: %d" count)))

(global-set-key (kbd "C-c h q") #'my/insert-hebrew-quotes)
(global-set-key (kbd "C-c h p") #'my/insert-hebrew-parens)
(global-set-key (kbd "C-c h c") #'my/count-hebrew-words)

(defun my/create-hebrew-document ()
  "Create a new Hebrew document."
  (interactive)
  (let* ((type (completing-read "Type: " '("org" "latex" "typst" "context")))
         (file (read-file-name (format "New %s file: " type))))
    (find-file file)
    (pcase type
      ("org" (insert (format "#+TITLE: כותרת
#+AUTHOR: שאול
#+DATE: %s
#+LATEX_CLASS: article-hebrew
" (format-time-string "%Y-%m-%d"))))
      ("latex" (insert "\\documentclass{article}
\\usepackage{fontspec}
\\usepackage[bidi=basic]{babel}
\\babelprovide[main, import]{hebrew}
\\babelfont{rm}{David CLM}

\\begin{document}

\\section{מבוא}

\\end{document}
"))
      ("typst" (insert "#set text(lang: \"he\", font: \"David CLM\")
// RTL is automatic from lang: he -- do NOT use page(flipped:true) (landscape)

= כותרת

"))
      ("context" (insert "\\mainlanguage[he]
\\setupalign[r2l]
\\definefontfamily[hebrew][rm][David CLM]

\\starttext

\\startsection[title={כותרת}]
\\stopsection

\\stoptext")))))

(global-set-key (kbd "C-c h n") #'my/create-hebrew-document)

(provide '00-hebrew)
