(defvar my/latex-rich-footnotes-preamble
  "% ══════════════════════════════════════════════════════════════
% RICH FOOTNOTES SYSTEM
% ══════════════════════════════════════════════════════════════

% Base: bigfoot for nesting
\\usepackage{bigfoot}
\\DeclareNewFootnote{default}
\\DeclareNewFootnote{B}
\\DeclareNewFootnote{C}
\\DeclareNewFootnote{D}
\\DeclareNewFootnote{E}
\\DeclareNewFootnote{F}
\\DeclareNewFootnote{G}
\\DeclareNewFootnote{H}
\\DeclareNewFootnote{I}
\\DeclareNewFootnote{J}

% Shortcuts
\\newcommand{\\rfn}[1]{\\footnote{#1}}
\\newcommand{\\rfnB}[1]{\\footnoteB{#1}}
\\newcommand{\\rfnC}[1]{\\footnoteC{#1}}

% Headings in footnotes
\\newcommand{\\fnh}[2]{%
  \\ifcase#1\\relax
  \\or\\par\\medskip{\\bfseries\\Large #2}\\par\\smallskip% 1
  \\or\\par\\smallskip{\\bfseries\\large #2}\\par\\smallskip% 2
  \\or\\par\\smallskip{\\bfseries #2}\\par\\smallskip% 3
  \\else\\par{\\bfseries #2}\\par% fallback
  \\fi
}

% Lists in footnotes
\\usepackage{enumitem}
\\setlistdepth{20}
\\newlist{fnitemize}{itemize}{20}
\\setlist[fnitemize]{nosep,leftmargin=*}

% Tables in footnotes
\\usepackage{tabularx}
\\newcounter{fntablecounter}
\\NewDocumentEnvironment{fntab}{m O{}}{%
  \\par\\smallskip
  \\refstepcounter{fntablecounter}%
  \\IfValueTF{#2}{\\noindent\\textbf{Table~\\thefntablecounter:} #2\\par\\smallskip}{}%
  \\centering\\begin{tabular}{#1}%
}{\\end{tabular}\\par\\smallskip}

% Code blocks
\\usepackage{listings}
\\NewDocumentEnvironment{fncode}{O{}}{%
  \\par\\smallskip\\lstset{language=#1}\\begin{lstlisting}%
}{\\end{lstlisting}\\par\\smallskip}
\\newcommand{\\fnc}[1]{\\texttt{#1}}"
  "LaTeX preamble for rich footnotes.")

(defvar my/typst-rich-footnotes-preamble
  "// Rich footnotes for Typst
#let fnh(level, body) = {
  let sizes = (1.3em, 1.2em, 1.1em, 1em)
  let idx = calc.min(level - 1, 3)
  text(size: sizes.at(idx), weight: \"bold\", body)
}

#let rfn(body) = footnote(body)

#let fn-table-counter = counter(\"fn-table\")
#let fn-table(caption: none, ..args) = {
  fn-table-counter.step()
  block[
    #if caption != none [*Table #fn-table-counter.display():* #caption #v(0.3em)]
    #table(..args)
  ]
}

#let fn-code(lang: none, body) = {
  block(fill: luma(245), inset: 5pt, radius: 2pt, raw(body.text, lang: lang))
}"
  "Typst preamble for rich footnotes.")

(defvar my/context-rich-footnotes-preamble
  "% Rich footnotes for ConTeXt
\\definenote[RFN]
\\definestartstop[startRFN][commands={\\startnote[RFN]},after={\\stopnote}]

\\define[2]\\fnh{%
  \\ifnum#1=1 \\blank[small]{\\bfc #2}\\blank[small]\\fi
  \\ifnum#1=2 \\blank[small]{\\bfb #2}\\blank[small]\\fi
  \\ifnum#1=3 \\blank[small]{\\bfa #2}\\blank[small]\\fi
}

\\setupitemize[each][packed]
\\definetyping[fncode][bodyfont=small\\tt]"
  "ConTeXt preamble for rich footnotes.")

;; ═══════════════════════════════════════════════════════════════════════════
;; ORG PREPROCESSOR
;; ═══════════════════════════════════════════════════════════════════════════

(defvar my/richfn-debug nil "Enable debug messages.")

(defun my/richfn--has-rich-content-p (text)
  "Check if TEXT contains rich org content."
  (and text
       (string-match-p
        (rx (or (seq bol (+ "*") " ")
                (seq bol (* space) (any "-+") " ")
                (seq bol (* space) (+ digit) (any ".)") " ")
                (seq bol (* space) "|")
                (seq bol (* space) "#+begin_src")))
        text)))

(defun my/richfn--org-to-latex (text)
  "Convert org TEXT to LaTeX."
  (with-temp-buffer
    (insert text)
    ;; Convert headings
    (goto-char (point-min))
    (while (re-search-forward "^\\(\\*+\\) \\(.*\\)$" nil t)
      (let ((level (min 20 (length (match-string 1))))
            (title (match-string 2)))
        (replace-match (format "\\\\fnh{%d}{%s}" level title) t t)))
    ;; Convert lists
    (goto-char (point-min))
    (while (re-search-forward "^\\([ \t]*\\)[-+] \\(.*\\)$" nil t)
      (replace-match "\\1\\\\item \\2" t nil))
    (goto-char (point-min))
    (while (re-search-forward "\\(\\\\item.*\n\\)+" nil t)
      (let ((items (match-string 0)))
        (replace-match (format "\\begin{fnitemize}\n%s\\end{fnitemize}\n" items))))
    (buffer-string)))

(defun my/richfn-preprocess-buffer (backend)
  "Preprocess rich footnotes before export."
  (when (memq backend '(latex beamer))
    (save-excursion
      (goto-char (point-min))
      (while (re-search-forward "^\\[fn:\\([^]]+\\)\\][ \t]*\n\\(.*?\\)\n\n" nil t)
        (let ((id (match-string 1))
              (content (match-string 2)))
          (when (my/richfn--has-rich-content-p content)
            (let ((converted (my/richfn--org-to-latex content)))
              (replace-match (format "[fn:%s] %s\n\n" id converted) t t))))))))

(with-eval-after-load 'ox
  (add-hook 'org-export-before-processing-hook #'my/richfn-preprocess-buffer))

(defun my/insert-rich-footnote-latex ()
  "Insert LaTeX rich footnote template."
  (interactive)
  (insert "\\rfn{%
  \\fnh{1}{Heading}
  
  \\begin{fnitemize}
    \\item First item
  \\end{fnitemize}
  
  \\begin{fntab}{|l|l|}[Caption]
    \\hline
    A & B \\\\
    \\hline
  \\end{fntab}
}")
  (search-backward "Heading"))

(defun my/insert-rich-footnote-typst ()
  "Insert Typst rich footnote template."
  (interactive)
  (insert "#rfn[
  #fnh(1)[Heading]
  
  - Item 1
  
  #fn-table(caption: \"Caption\", columns: 2, [A], [B])
]")
  (search-backward "Heading"))

(defun my/insert-rich-footnote-context ()
  "Insert ConTeXt rich footnote template."
  (interactive)
  (insert "\\startRFN
\\fnh{1}{Heading}

\\startitemize[packed]
\\item First item
\\stopitemize
\\stopRFN")
  (search-backward "Heading"))
