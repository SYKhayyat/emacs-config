* Local AI with Ollama

#+begin_src emacs-lisp
;;; 21-local-ai.el --- Local AI Research Assistant -*- lexical-binding: t; -*-

(use-package gptel
  :bind (("C-c g g" . gptel)
         ("C-c g s" . gptel-send)
         ("C-c g m" . gptel-menu))
  :config
  ;; Define the local Ollama backend
  (setq gptel-backend 
        (gptel-make-ollama "Ollama"
          :host "localhost:11434"
          :stream t
          :models '("llama3" "mistral")))

  ;; Default to Org-mode for AI responses
  (setq gptel-default-mode 'org-mode)
  
  ;; Custom scholar-focused prompt
  (setq gptel-system-prompt 
        "You are a sophisticated research assistant for a scholar of Jewish texts. 
         You are expert in Hebrew, Aramaic, and English. 
         You provide precise translations, logical analysis of legal arguments, 
         and clear summaries of complex philosophical concepts."))

(provide '21-local-ai)
;;; 21-local-ai.el ends here
#+end_src
